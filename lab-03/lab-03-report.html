const express = require('express');
const path = require('path');
const { open } = require('sqlite');
const sqlite3 = require('sqlite3');

const app = express();
app.use(express.json());

const dbPath = path.join(__dirname, 'books.db');
let db = null;
const PORT = process.env.PORT || 3000;

// Initialize SQLite connection and boot server
const initializeDBAndServer = async () => {
  try {
    db = await open({ filename: dbPath, driver: sqlite3.Database });
    app.listen(PORT, () => console.log(`Server running at http://localhost:${PORT}/`));
  } catch (error) {
    console.error(`DB Error: ${error.message}`);
    process.exit(1);
  }
};
initializeDBAndServer();

// GET /books - Search, Sorting, Filtering, and Pagination
app.get('/books', async (req, res) => {
  const { search_q = '', order_by = 'id', order = 'ASC', limit = 10, offset = 0, min_price = 0, max_price = 1000000 } = req.query;
  const allowedCols = ['id', 'title', 'author', 'price', 'pages'];
  const validOrderBy = allowedCols.includes(order_by) ? order_by : 'id';
  const validOrder = order.toUpperCase() === 'DESC' ? 'DESC' : 'ASC';

  const query = `
    SELECT * FROM book
    WHERE title LIKE ? AND price BETWEEN ? AND ?
    ORDER BY ${validOrderBy} ${validOrder}
    LIMIT ? OFFSET ?;
  `;
  const books = await db.all(query, [`%${search_q}%`, min_price, max_price, limit, offset]);
  res.status(200).json(books);
});

// POST /books - Validation & 201 Response
app.post('/books', async (req, res) => {
  const { title, author, price, pages } = req.body;
  if (!title || typeof title !== 'string' || title.trim() === '') {
    return res.status(400).json({ message: "Validation Error: 'title' is required." });
  }
  if (!author || typeof author !== 'string' || author.trim() === '') {
    return res.status(400).json({ message: "Validation Error: 'author' is required." });
  }
  if (price === undefined || isNaN(Number(price)) || Number(price) < 0) {
    return res.status(400).json({ message: "Validation Error: 'price' must be a non-negative number." });
  }

  const result = await db.run(
    `INSERT INTO book (title, author, price, pages) VALUES (?, ?, ?, ?);`,
    [title.trim(), author.trim(), Number(price), Number(pages) || 100]
  );
  res.status(201).json({ message: 'Book Added Successfully', bookId: result.lastID });
});

4. Debugging Exercise Walkthrough

Bug 1: Port Conflict (EADDRINUSE)
Issue: Server crashes on launch with Error: listen EADDRINUSE: address already in use :::3000 when another process or previous run occupies the port.
Fix: Parameterized the port with environment fallback: const PORT = process.env.PORT || 3000; and killed active process using PowerShell PID search.
Bug 2: Wrong Import Path
Issue: Runtime error Cannot find module './database' due to invalid relative resolution.
Fix: Replaced faulty module imports with direct sqlite destructuring and path.join(__dirname, 'books.db').
Bug 3: Table-Name Typo
Issue: Queries failed with SQLITE_ERROR: no such table: books because DDL created book (singular) while queries targeted books (plural).
Fix: Standardized table reference to book consistently across all queries and schemas.
Bug 4: Missing `await` Keyword
Issue: API returned pending Promises {} or crashed due to unhandled async results.
Fix: Ensured all database calls (db.all, db.get, db.run) are awaited within async Express route handlers.

6. Viva Voce Questions & Detailed Answers

Q1: Query Parameter vs. Path Parameter?
Path parameters (e.g., /books/:bookId) are mandatory parts of the URL path used to identify a distinct individual resource entity. Query parameters (e.g., /books?order_by=price&limit=5) appear after the ? delimiter as optional key-value pairs used to modify, filter, paginate, or sort the resource collection without identifying a different resource type.
Q2: What does "stateless" mean in REST architecture?
A stateless REST API does not retain any client session context on the server between requests. Every incoming HTTP request must encapsulate all necessary data (authentication credentials, parameters, payload) required for the server to process it completely and independently.
Q3: Which status code is used for bad input vs. a missing record?
400 Bad Request is used for invalid or malformed client input (such as omitted mandatory fields or non-numeric types). 404 Not Found is returned when the request format is valid, but the requested URI / resource ID does not exist in the database.
Q4: Why await the database call and use parameterized queries?
Database calls are asynchronous I/O operations; using await ensures the query resolves before the response is dispatched to the client. Parameterized queries (? placeholders) separate the SQL execution logic from user-provided data, rendering SQL injection impossible because inputs are treated strictly as data literals.
